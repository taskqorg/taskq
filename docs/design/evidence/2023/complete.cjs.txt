'use strict';
const fs=require('node:fs'),path=require('node:path'),assert=require('node:assert/strict'),p=require('./measure.cjs'),base=__dirname;

function browser(method,params={}){return new Promise((resolve,reject)=>{const id=++serial,t=setTimeout(()=>{pending.delete(id);reject(Error(method+' browser timeout'));},15000);pending.set(id,{resolve,reject,t});ws.send(JSON.stringify({id,method,params}));});}
async function main(){

  await p.until('document.querySelector("[data-c=latest]")?.innerText.includes("New messages")');await p.settle();c.append_delta=await p.displacement(c.before);await p.delay(150);c.delayed_delta=await p.displacement(c.before);
  await p.ev('document.querySelector("[data-c=latest]").click()');await p.until('document.querySelector(".dialogue-transcript").innerText.includes('+JSON.stringify('Final native append '+rt)+')');await p.settle();
  c.measures=await p.ev('({renders:measure.renders.splice(0),longtasks:measure.longtasks.splice(0)})');save();
 }
 if(result.trace.state==='recording'){
  try{
   await browser('Tracing.end');const end=Date.now()+20000;while(!traceReceipt&&Date.now()<end)await p.delay(50);assert(traceReceipt?.stream,'browser trace stream receipt missing');
   let data='';for(;;){const r=await browser('IO.read',{handle:traceReceipt.stream,size:1024*1024});data+=r.base64Encoded?Buffer.from(r.data,'base64').toString('utf8'):r.data;if(r.eof)break;}await browser('IO.close',{handle:traceReceipt.stream});
   const events=JSON.parse(data).traceEvents,names=new Map(events.filter(e=>e.name==='thread_name').map(e=>[e.pid+':'+e.tid,e.args?.name]));
   result.trace={state:'complete',event_count:events.length,tasks:events.filter(e=>e.ph==='X'&&['RunTask','ThreadControllerImpl::RunTask','FunctionCall','EventDispatch','FireAnimationFrame','UpdateLayoutTree','Layout','Paint'].includes(e.name)).map(e=>({name:e.name,thread:names.get(e.pid+':'+e.tid)||'unknown',ts_us:e.ts,duration_ms:e.dur/1000})),marks:events.filter(e=>/^(prepend|append)-/.test(e.name)).map(e=>({name:e.name,ts_us:e.ts}))};
  }catch(e){result.trace={state:'unknown',reason:e.message};}
 }
 ws.close();save();
 await p.choose('claude-normal');await p.ev('(()=>{const e=document.querySelector(".conversation-reader");e.scrollTop=e.scrollHeight*.4;e.dispatchEvent(new Event("scroll",{bubbles:true}));})()');await p.settle();await p.delay(400);
 result.restart={before:await p.anchor(),navigation:JSON.parse(fs.readFileSync(path.join(base,'navigation.json'),'utf8'))};save();await p.stop();

 result.finished=new Date().toISOString();save();console.log('DONE',result.trace.state,result.restart.delta);
}
main().catch(async e=>{result.error=e.message;save();console.error(e.stack);try{ws?.close();await p.stop();}catch{}process.exitCode=1;});
